function afficherImagesProcedure(reponse) { if ( !reponse || !Array.isArray(reponse.images) || reponse.images.length === 0 ) { return ""; } const images = reponse.images .map((image) => { const url = valeurTexte( image?.url || image?.image, "" ); if (!url) { return ""; } const pageManuel = valeurTexte( image?.page_manuel, "—" ); const pagePDF = valeurTexte( image?.page_pdf, "—" ); return `

Illustration de la procédure

Manuel ${echapperHTML( pageManuel )} · PDF ${echapperHTML( pagePDF )}

Page source du Service Manual utilisée pour les opérations de dépose.

`; }) .join(""); if (!images) { return ""; } return `

Illustration du Service Manual

Les opérations graphiques de dépose sont visibles directement sur la page source du manuel.

${images}
`; }
